21bool DevLab_MPU6050::begin(uint8_t sdaPin, uint8_t sclPin, uint32_t clock) {
22 _wire->begin(sdaPin, sclPin);
23 _wire->setClock(clock);
24
25 delay(50);
26
27 uint8_t whoAmI = WhoAmI();
28 if (whoAmI != 0x68) {
29 return false;
30 }
31
32 writeRegister8(REG_PWR_MGMT_1, 0x01); // Wake up the MPU6050
33 delay(10);
34
35 writeRegister8(REG_SMPLRT_DIV, 0x07);
36 writeRegister8(REG_CONFIG, 0x06);
37 writeRegister8(REG_GYRO_CONFIG, 0x08);
38 writeRegister8(REG_ACCEL_CONFIG, 0x10);
39
40 return true;
41}

47bool DevLab_MPU6050::readData(SensorData &dataOut) {
48 _wire->beginTransmission(_address);
49 _wire->write(REG_ACCEL_XOUT_H);
50 if (_wire->endTransmission(false) != 0) {
51 return false;
52 }
53
54 _wire->requestFrom(_address, (uint8_t)14, (uint8_t)true);
55
56 if (_wire->available() < 14) {
57 return false;
58 }
59
60 dataOut.accX = (int16_t)(_wire->read() << 8 | _wire->read());
61 dataOut.accY = (int16_t)(_wire->read() << 8 | _wire->read());
62 dataOut.accZ = (int16_t)(_wire->read() << 8 | _wire->read());
63
64 dataOut.tempRaw = (int16_t)(_wire->read() << 8 | _wire->read());
65 dataOut.temperatureC = dataOut.tempRaw / 340.0f + 36.53f;
66
67 dataOut.gyrX = (int16_t)(_wire->read() << 8 | _wire->read());
68 dataOut.gyrY = (int16_t)(_wire->read() << 8 | _wire->read());
69 dataOut.gyrZ = (int16_t)(_wire->read() << 8 | _wire->read());
70
71 return true;
72}

156uint8_t DevLab_MPU6050::readRegister8(uint8_t reg) {
157 _wire->beginTransmission(_address);
158 _wire->write(reg);
159 _wire->endTransmission(false);
160
161 _wire->requestFrom(_address, (uint8_t)1, (uint8_t)true);
162 if (_wire->available() < 1) {
163 return 0;
164 }
165
166 return _wire->read();
167}
