24const uint8_t ACCEL_SCALES[] = {0, 1, 2, 3};
25const uint8_t GYRO_SCALES[] = {0, 1, 2, 3};
26const uint8_t DLPF_SETTINGS[] = {0, 1, 2, 3, 4, 5, 6};
27
28// Element counts computed explicitly -- sizeof(array) on a uint8_t[]
29// happens to equal the element count only because uint8_t is 1 byte.
30const uint8_t N_ACCEL = sizeof(ACCEL_SCALES) / sizeof(ACCEL_SCALES[0]);
31const uint8_t N_GYRO = sizeof(GYRO_SCALES) / sizeof(GYRO_SCALES[0]);
32const uint8_t N_DLPF = sizeof(DLPF_SETTINGS) / sizeof(DLPF_SETTINGS[0]);
33
34const uint8_t CLK_SOURCE_PLL_GYRO_X = 1; // stable reference, recommended by Invensense
35
36// DLPF_CFG bandwidths in Hz (MPU-6000/6050 register map, DLPF_CFG table).
37const float ACCEL_BW_HZ[] = {260.0f, 184.0f, 94.0f, 44.0f, 21.0f, 10.0f, 5.0f};
38const float GYRO_BW_HZ[] = {256.0f, 188.0f, 98.0f, 42.0f, 20.0f, 10.0f, 5.0f};
39
40// Nyquist must be at least this many times the DLPF corner frequency.
41const float NYQUIST_MARGIN = 1.2f;
42
43// Number of sample periods to wait after an AFS_SEL/FS_SEL change before
44// trusting readData() -- absorbs I2C transaction latency and ADC
45// conversion jitter so we don't sample mid-transition.
46const uint8_t SCALE_SETTLE_PERIODS = 5;
47
48// How many full sweeps to run back-to-back (set from setup(), or change
49// here and call sweepAndPrint() once per iteration from setup()/loop()).
50const uint16_t N_REPEATS = 100;
51
52// Internal sampling rate feeding SMPLRT_DIV: 8kHz when the DLPF is
53// bypassed (DLPF_CFG=0), 1kHz for any other DLPF setting.
54float internalFsHz(uint8_t dlpf) {
55 return (dlpf == 0) ? 8000.0f : 1000.0f;
56}
57
58// SMPLRT_DIV per DLPF_CFG, computed once at startup. Chosen as the
59// LARGEST divisor (lowest output rate, gentlest on I2C traffic) that
60// still keeps Nyquist >= NYQUIST_MARGIN * corner.
61uint8_t g_smplrtDivByDlpf[7];
62
63void computeSmplrtDivTable() {
64 for (uint8_t d = 0; d < N_DLPF; d++) {
65 uint8_t dlpf = DLPF_SETTINGS[d];
66 float fsBase = internalFsHz(dlpf);
67 float cornerHz = min(ACCEL_BW_HZ[d], GYRO_BW_HZ[d]);
68
69 float maxRatio = fsBase / (2.0f * NYQUIST_MARGIN * cornerHz);
70 long divCandidate = (long)floor(maxRatio - 1.0f);
71
72 if (divCandidate < 0) divCandidate = 0;
73 if (divCandidate > 255) divCandidate = 255;
74
75 g_smplrtDivByDlpf[d] = (uint8_t)divCandidate;
76 }
77}
78
79float samplePeriodMs(uint8_t dIndex) {
80 uint8_t dlpf = DLPF_SETTINGS[dIndex];
81 float fsBase = internalFsHz(dlpf);
82 uint8_t div = g_smplrtDivByDlpf[dIndex];
83 return 1000.0f * (1 + div) / fsBase;
84}
85
86// Time for the DLPF's single-pole low-pass to settle (~5 time constants,
87// tau = 1/(2*pi*BW)), using the tighter of the accel/gyro corner
88// frequencies, plus one sample period so the settled value has actually
89// been latched into the output registers.
90uint16_t dlpfSettleMs(uint8_t dIndex) {
91 float bwHz = min(ACCEL_BW_HZ[dIndex], GYRO_BW_HZ[dIndex]);
92 float tauMs = 1000.0f / (2.0f * PI * bwHz);
93 float settleMs = 5.0f * tauMs + samplePeriodMs(dIndex);
94 return (uint16_t)ceil(settleMs);
95}
96
97// Time to wait after an AFS_SEL/FS_SEL change: several sample periods.
98uint16_t scaleSettleMs(uint8_t dIndex) {
99 float settleMs = SCALE_SETTLE_PERIODS * samplePeriodMs(dIndex);
100 return (uint16_t)ceil(settleMs);
101}

103void sweepAndPrint() {
104 // --- Repeat-call fix -------------------------------------------------
105 // Explicitly (re)apply and settle the FIRST combination of the sweep
106 // (AFS_SEL=0, FS_SEL=0, DLPF_CFG=0) before entering the nested loops.
107 // Without this, if sweepAndPrint() is called back-to-back in a loop,
108 // this first reading follows immediately after the LAST combination
109 // of the previous call (AFS_SEL=3, DLPF_CFG=6) with no settle at all,
110 // and can carry over the previous call's scale (see header note).
111 // dlpfSettleMs() is used here (not scaleSettleMs()) because it's the
112 // longer of the two and covers both the DLPF and the scale settling.
113 imu.setDLPF(DLPF_SETTINGS[0]);
114 imu.sampleRateDiv(g_smplrtDivByDlpf[0]);
115 imu.setAccelScale(ACCEL_SCALES[0]);
116 imu.setGyroScale(GYRO_SCALES[0]);
117 delay(dlpfSettleMs(0));
118 // ----------------------------------------------------------------------
119
120 for (uint8_t d = 0; d < N_DLPF; d++) {
121 uint8_t dlpf = DLPF_SETTINGS[d];
122 uint8_t div = g_smplrtDivByDlpf[d];
123
124 imu.setDLPF(dlpf);
125 imu.sampleRateDiv(div);
126 delay(dlpfSettleMs(d));
127
128 float fsBase = internalFsHz(dlpf);
129 float outputRateHz = fsBase / (1 + div);
130 float nyquistHz = outputRateHz / 2.0f;
131 float gyroBw = GYRO_BW_HZ[d];
132 float cornerHz = min(ACCEL_BW_HZ[d], GYRO_BW_HZ[d]);
133 bool belowNyquist = (cornerHz <= nyquistHz);
134
135 uint16_t scaleDelayMs = scaleSettleMs(d);
136
137 for (uint8_t a = 0; a < N_ACCEL; a++) {
138 imu.setAccelScale(ACCEL_SCALES[a]);
139
140 for (uint8_t g = 0; g < N_GYRO; g++) {
141 imu.setGyroScale(GYRO_SCALES[g]);
142
143 delay(scaleDelayMs);
144
145 Serial.print(ACCEL_SCALES[a]); Serial.print(',');
146 Serial.print(GYRO_SCALES[g]); Serial.print(',');
147 Serial.print(dlpf); Serial.print(',');
148 Serial.print(div); Serial.print(',');
149
150 if (imu.readData(data)) {
151 Serial.print(data.accX); Serial.print(',');
152 Serial.print(data.accY); Serial.print(',');
153 Serial.print(data.accZ); Serial.print(',');
154 Serial.print(data.gyrX); Serial.print(',');
155 Serial.print(data.gyrY); Serial.print(',');
156 Serial.print(data.gyrZ); Serial.print(',');
157 Serial.print(data.temperatureC, 2); Serial.print(',');
158
159 Serial.print(gyroBw, 1); Serial.print(',');
160 Serial.print(outputRateHz, 2); Serial.print(',');
161 Serial.print(nyquistHz, 2); Serial.print(',');
162 Serial.println(belowNyquist ? "OK" : "ALIAS_RISK");
163 } else {
164 Serial.println("READ_ERROR");
165 }
166 }
167 }
168 }
169}

171void setup() {
172 Serial.begin(115200);
173 delay(1000);
174
175 Serial.println("Inicializando MPU6050...");
176
177 if (!imu.begin(SDA_PIN, SCL_PIN, 400000)) {
178 Serial.println("ERROR: MPU6050 no encontrado.");
179 while (1) {
180 delay(1000);
181 }
182 }
183
184 imu.setClkSource(CLK_SOURCE_PLL_GYRO_X);
185 computeSmplrtDivTable();
186
187 Serial.println("MPU6050 inicializado correctamente.");
188 Serial.println("Tabla SMPLRT_DIV calculada por DLPF_CFG (Nyquist >= 1.2x corner):");
189 for (uint8_t d = 0; d < N_DLPF; d++) {
190 Serial.print("# DLPF_CFG=");
191 Serial.print(DLPF_SETTINGS[d]);
192 Serial.print(" -> SMPLRT_DIV=");
193 Serial.println(g_smplrtDivByDlpf[d]);
194 }
195
196 Serial.print("Repeticiones del sweep completo: ");
197 Serial.println(N_REPEATS);
198
199 Serial.println("Iniciando sweep de configuraciones...");
200 Serial.println("accelScale,gyroScale,dlpf,smplrtDiv,accX,accY,accZ,gyrX,gyrY,gyrZ,tempC,gyroBW_Hz,outputRate_Hz,nyquist_Hz,dlpf_below_nyquist");
201
202 for (uint16_t i = 0; i < N_REPEATS; i++) {
203 sweepAndPrint();
204 }
205
206 Serial.println("Sweep completo.");
207}
